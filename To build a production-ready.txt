def generate_contextual_vibe(local_time: str, weather: str, location: str, activity_bpm: int) -> dict:
    """
    Uses the Groq LLM to act as the AI DJ. 
    It analyzes context and returns a JSON payload with a greeting and target music genres.
    """
    prompt = f"""
    You are 'Vespera', an advanced, emotionally intelligent AI Music DJ. 
    Analyze the user's current context:
    - Time: {local_time}
    - Weather: {weather}
    - Location: {location}
    - User Activity: {activity_bpm} BPM (If > 100, they are exercising/running. If < 60, they are resting).

    Based on this, determine the perfect musical vibe.
    Return ONLY a valid JSON object with the following structure:
    {{
        "greeting": "A short, cool UI greeting like 'Late night in Jabalpur' or 'Rainy Morning Focus'",
        "dj_message": "A 1-sentence friendly message explaining why you chose this vibe.",
        "search_query": "3-4 keywords to search in our Vector Database (e.g., 'high energy cyberpunk synthwave', 'lofi acoustic acoustic chuva')",
        "target_bpm": The ideal BPM for the music (integer)
    }}
    """

    try:
        # We enforce JSON response format so the API never breaks
        response = client.chat.completions.create(
            messages=[{"role": "user", "content": prompt}],
            model="llama3-8b-8192",
            temperature=0.7,
            response_format={"type": "json_object"} 
        )
        return json.loads(response.choices[0].message.content)
    except Exception as e:
        print(f"LLM Error: {e}")
        # Fallback if API fails
        return {
            "greeting": "Welcome to Vespera",
            "dj_message": "Curating your personal soundscape...",
            "search_query": "popular trending music",
            "target_bpm": 100
        }

import requests
from pinecone import Pinecone
from app.config import settings

# Initialize Pinecone
pc = Pinecone(api_key=settings.PINECONE_API_KEY)

def get_huggingface_embedding(text: str) -> list:
    """
    Uses HuggingFace's free inference API to convert text to vectors.
    Saves you from running heavy embedding models on your free Render server.
    """
    model_id = "sentence-transformers/all-MiniLM-L6-v2"
    api_url = f"https://api-inference.huggingface.co/pipeline/feature-extraction/{model_id}"
    headers = {"Authorization": f"Bearer {settings.HF_TOKEN}"}
    
    response = requests.post(api_url, headers=headers, json={"inputs": [text]})
    if response.status_code == 200:
        return response.json()[0]
    return [0.0] * 384 # Fallback empty vector

def search_music(query: str, target_bpm: int, limit: int = 4):
    """
    Searches Pinecone Vector DB for songs matching the AI's vibe.
    """
    try:
        index = pc.Index(settings.PINECONE_INDEX_NAME)
        query_vector = get_huggingface_embedding(query)
        
        # Query Pinecone using the vector AND metadata filtering (BPM)
        results = index.query(
            vector=query_vector,
            top_k=limit,
            include_metadata=True,
            # If user is running, only show fast songs.
            filter={"bpm": {"$gte": target_bpm - 10, "$lte": target_bpm + 10}} if target_bpm > 100 else None
        )
        
        # Format for Flutter
        playlists = []
        for match in results['matches']:
            meta = match['metadata']
            playlists.append({
                "title": meta.get("title", "Unknown Track"),
                "subtitle": f"{meta.get('genre', 'Vibe')} • {meta.get('bpm', '')} BPM",
                "image_url": meta.get("image_url", "https://images.unsplash.com/photo-1618005182384-a83a8bd57fbe")
            })
        return playlists

    except Exception as e:
        print(f"Pinecone Error/Empty DB: {e}")
        # IMPORTANT: Fallback mock data so your app works IMMEDIATELY before you populate Pinecone
        return _mock_data(query)

Yes, we absolutely did use Deep Learning, but in a very specific, highly optimized way.
In industry-standard applications, Deep Learning is split into two phases: Inference (Real-time) and Batch Processing (Offline).
Here is exactly where and how Deep Learning is working in the architecture we just built:
1. Deep Learning in the Real-Time API (What we just wrote)
In the pinecone_db.py file, we used this function:
https://api-inference.huggingface.co/pipeline/feature-extraction/sentence-transformers/all-MiniLM-L6-v2
What this is: all-MiniLM-L6-v2 is a Deep Learning Transformer model (specifically a dense neural network).
What it does: It takes human language and uses deep neural layers to convert it into a 384-dimensional mathematical array (a vector). This is the core of the RAG engine.
The LLM: Llama 3 (running on Groq) is, of course, a massive Deep Learning neural network containing 8 billion parameters.
2. Deep Learning for Audio (The Offline Step)
You might be wondering: "Wait, where is the code that actually listens to the music and calculates the BPM, pitch, and energy using Librosa like we talked about?"
That part is intentionally kept out of the FastAPI server code.
If you try to run heavy audio Deep Learning (scanning a 4-minute MP3 file) inside a live web API, it will take 10 seconds per song. Your app would freeze, and Render.com would crash your free server because you ran out of RAM.
Instead, companies like Spotify do this offline. You run a separate Python script on your laptop once to analyze the MP3s using Deep Learning, and then you push that math into Pinecone. The FastAPI backend just reads the results.

import librosa
import numpy as np
from pinecone import Pinecone
import requests

# Your Free API Keys
HF_TOKEN = "your_huggingface_token"
PC_API_KEY = "your_pinecone_key"

pc = Pinecone(api_key=PC_API_KEY)
index = pc.Index("vespera-index")

def extract_audio_features(mp3_path):
    """
    DEEP LEARNING / DSP MACHINE LEARNING:
    Uses Librosa to mathematically analyze the actual audio waveform.
    """
    print(f"Analyzing {mp3_path}...")
    # Load the audio file (y = audio time series, sr = sampling rate)
    y, sr = librosa.load(mp3_path, duration=60) # Analyze first 60 seconds
    
    # 1. Calculate Tempo (BPM)
    tempo, _ = librosa.beat.beat_track(y=y, sr=sr)
    
    # 2. Calculate Energy (Root Mean Square)
    rms = librosa.feature.rms(y=y)
    energy = float(np.mean(rms))

    return {"bpm": round(tempo[0]), "energy": energy}

def embed_and_upload(song_title, genre, mp3_path, image_url):
    # 1. Run the Audio Deep Learning analysis
    audio_math = extract_audio_features(mp3_path)
    
    # 2. Create the text description for the LLM to search
    text_to_embed = f"Song: {song_title}. Genre: {genre}. Vibe: High energy {genre}."
    
    # 3. Get the Text Deep Learning Embedding (HuggingFace)
    api_url = "https://api-inference.huggingface.co/pipeline/feature-extraction/sentence-transformers/all-MiniLM-L6-v2"
    headers = {"Authorization": f"Bearer {HF_TOKEN}"}
    vector = requests.post(api_url, headers=headers, json={"inputs": [text_to_embed]}).json()[0]

    # 4. Upload BOTH to Pinecone DB
    index.upsert(
        vectors=[
            {
                "id": song_title.lower().replace(" ", "_"), 
                "values": vector, 
                "metadata": {
                    "title": song_title,
                    "genre": genre,
                    "bpm": audio_math["bpm"],
                    "energy": audio_math["energy"],
                    "image_url": image_url
                }
            }
        ]
    )
    print(f"Successfully uploaded {song_title} to Vector Database!")

# --- Run it ---

final AudioPlayer _audioPlayer = AudioPlayer();
final YoutubeExplode _yt = YoutubeExplode();

Future<void> playSong(String searchQuery) async {
  try {
    // 1. Silently search YouTube for the audio
    var searchResults = await _yt.search.search("$searchQuery official audio");
    var video = searchResults.first;

    // 2. Extract the highest quality audio-only stream
    var manifest = await _yt.videos.streamsClient.getManifest(video.id);
    var audioStream = manifest.audioOnly.withHighestBitrate();

    // 3. Play it in the background!
    await _audioPlayer.setUrl(audioStream.url.toString());
    _audioPlayer.play();
    
    print("Now Playing: ${video.title}");
  } catch (e) {
    print("Error playing audio: $e");
  }
}
